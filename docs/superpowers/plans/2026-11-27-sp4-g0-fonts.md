```python
"""SP4 G0: the subtitle font registry is the one list of fonts (spec 5.2)."""
import hashlib
import json
import struct
from pathlib import Path

FONTS = Path(__file__).resolve().parents[1] / "fonts"
REGISTRY_PATH = FONTS / "fonts.json"

EXPECTED_FAMILIES = {
    "Arimo", "Cousine", "Anton", "Montserrat", "Roboto",
    "Oswald", "Poppins", "Bebas Neue", "Permanent Marker",
}
SINGLE_WEIGHT = {"Anton", "Bebas Neue", "Permanent Marker"}


def _registry() -> dict:
    return json.loads(REGISTRY_PATH.read_text(encoding="utf-8"))


def _tables(data: bytes) -> dict:
    count = struct.unpack(">H", data[4:6])[0]
    out = {}
    for i in range(count):
        tag, _checksum, offset, length = struct.unpack(">4sIII", data[12 + 16 * i:28 + 16 * i])
        out[tag.decode("latin-1")] = (offset, length)
    return out


def _face_facts(path: Path) -> dict:
    """Family (name id 1), style (2), PostScript name (6) and weight class, without fontTools."""
    data = path.read_bytes()
    tables = _tables(data)
    name_off, _ = tables["name"]
    _fmt, count, string_off = struct.unpack(">HHH", data[name_off:name_off + 6])
    names = {}
    for i in range(count):
        rec = data[name_off + 6 + 12 * i:name_off + 18 + 12 * i]
        platform, _enc, lang, name_id, length, offset = struct.unpack(">6H", rec)
        if platform == 3 and lang == 0x409 and name_id in (1, 2, 6):
            start = name_off + string_off + offset
            names[name_id] = data[start:start + length].decode("utf-16-be")
    os2_off, _ = tables["OS/2"]
    weight = struct.unpack(">H", data[os2_off + 4:os2_off + 6])[0]
    return {"family": names.get(1), "style": names.get(2), "postscript": names.get(6),
            "weight": weight, "variable": "fvar" in tables}


def _faces():
    for fam in _registry()["families"]:
        for weight in ("regular", "bold"):
            if fam[weight]:
                yield fam, weight, fam[weight]


def test_registry_lists_the_promised_families_and_default():
    reg = _registry()
    assert {f["family"] for f in reg["families"]} == EXPECTED_FAMILIES
    assert reg["default"] == "Arimo"
    aliases = {a: f["family"] for f in reg["families"] for a in f["aliases"]}
    assert aliases == {"Arial": "Arimo", "Courier New": "Cousine", "Impact": "Anton"}
    for fam in reg["families"]:
        assert (fam["bold"] is None) == (fam["family"] in SINGLE_WEIGHT), fam["family"]


def test_every_face_file_exists_and_matches_its_hash():
    for fam, _weight, face in _faces():
        path = FONTS / face["file"]
        assert path.is_file(), face["file"]
        assert hashlib.sha256(path.read_bytes()).hexdigest() == face["sha256"], face["file"]


def test_every_face_is_static_and_named_as_the_registry_says():
    for fam, weight, face in _faces():
        facts = _face_facts(FONTS / face["file"])
        assert facts["family"] == fam["family"], face["file"]
        assert facts["postscript"] == face["postscript"], face["file"]
        assert facts["style"] == ("Bold" if weight == "bold" else "Regular"), face["file"]
        assert facts["weight"] == (700 if weight == "bold" else 400), face["file"]
        assert not facts["variable"], face["file"]


def test_each_family_ships_its_license_file():
    for fam in _registry()["families"]:
        lic = FONTS / fam["license"]
        assert lic.is_file() and lic.stat().st_size > 1000, fam["license"]


def test_no_font_file_outside_the_registry():
    listed = {face["file"] for _f, _w, face in _faces()}
    assert {p.name for p in FONTS.glob("*.ttf")} == listed
```

```python
"""Builds backend/fonts from pinned Google Fonts sources (SP4 spec 5.2).

Developer tool, never run by the app. Needs network access and
`pip install -r scripts/requirements-fonts.txt`.

    python scripts/build_fonts.py

Variable fonts are cut into static Regular (400) and Bold (700) files with
explicit names, because libass picks a face by family + weight and fakes bold
when only a Regular file exists. Output is deterministic (no timestamp update).
"""
import hashlib
import io
import json
import sys
import urllib.request
from pathlib import Path

from fontTools.ttLib import TTFont
from fontTools.varLib import instancer

COMMIT = "5026800714d07231937d92eb4cceabf06e05d74e"
RAW = f"https://raw.githubusercontent.com/google/fonts/{COMMIT}/"
ROOT = Path(__file__).resolve().parents[1]
OUT = ROOT / "backend" / "fonts"
DEFAULT = "Arimo"

# "variable": cut 400 and 700 from this file. "static": copy these files as they are.
FAMILIES = [
    {"family": "Arimo", "dir": "ofl/arimo", "license": "OFL.txt", "aliases": ["Arial"],
     "variable": "Arimo[wght].ttf"},
    {"family": "Cousine", "dir": "ofl/cousine", "license": "OFL.txt", "aliases": ["Courier New"],
     "static": {"regular": "Cousine-Regular.ttf", "bold": "Cousine-Bold.ttf"}},
    {"family": "Anton", "dir": "ofl/anton", "license": "OFL.txt", "aliases": ["Impact"],
     "static": {"regular": "Anton-Regular.ttf"}},
    {"family": "Montserrat", "dir": "ofl/montserrat", "license": "OFL.txt", "aliases": [],
     "variable": "Montserrat[wght].ttf"},
    {"family": "Roboto", "dir": "ofl/roboto", "license": "OFL.txt", "aliases": [],
     "variable": "Roboto[wdth,wght].ttf", "fixed_axes": {"wdth": 100}},
    {"family": "Oswald", "dir": "ofl/oswald", "license": "OFL.txt", "aliases": [],
     "variable": "Oswald[wght].ttf"},
    {"family": "Poppins", "dir": "ofl/poppins", "license": "OFL.txt", "aliases": [],
     "static": {"regular": "Poppins-Regular.ttf", "bold": "Poppins-Bold.ttf"}},
    {"family": "Bebas Neue", "dir": "ofl/bebasneue", "license": "OFL.txt", "aliases": [],
     "static": {"regular": "BebasNeue-Regular.ttf"}},
    {"family": "Permanent Marker", "dir": "apache/permanentmarker", "license": "LICENSE.txt", "aliases": [],
     "static": {"regular": "PermanentMarker-Regular.ttf"}},
]


def fetch(path: str) -> bytes:
    url = RAW + path.replace("[", "%5B").replace("]", "%5D")
    req = urllib.request.Request(url, headers={"User-Agent": "AutoClipper-FontBuild"})
    with urllib.request.urlopen(req, timeout=60) as resp:
        return resp.read()


def set_static_names(font: TTFont, family: str, style: str) -> None:
    ps_family = family.replace(" ", "")
    name = font["name"]
    name.removeNames(platformID=1)
    for name_id in (16, 17, 21, 22, 25):
        name.removeNames(nameID=name_id)
    full = family if style == "Regular" else f"{family} {style}"
    values = {1: family, 2: style, 3: f"{ps_family}-{style};{COMMIT[:7]}", 4: full, 6: f"{ps_family}-{style}"}
    for name_id, value in values.items():
        name.setName(value, name_id, 3, 1, 0x409)
    os2 = font["OS/2"]
    os2.usWeightClass = 700 if style == "Bold" else 400
    os2.fsSelection &= ~((1 << 0) | (1 << 5) | (1 << 6))
    os2.fsSelection |= (1 << 5) if style == "Bold" else (1 << 6)
    font["head"].macStyle = 1 if style == "Bold" else 0
    if "STAT" in font:
        del font["STAT"]


def save(font: TTFont, path: Path) -> None:
    font.recalcTimestamp = False
    font.save(str(path))


def postscript_name(path: Path) -> str:
    return TTFont(str(path))["name"].getDebugName(6)


def face_entry(path: Path) -> dict:
    return {"file": path.name, "postscript": postscript_name(path),
            "sha256": hashlib.sha256(path.read_bytes()).hexdigest()}


def build_family(spec: dict) -> dict:
    family = spec["family"]
    ps_family = family.replace(" ", "")
    faces = {"regular": None, "bold": None}
    if "variable" in spec:
        source = fetch(f"{spec['dir']}/{spec['variable']}")
        for weight, style in (("regular", "Regular"), ("bold", "Bold")):
            axes = {"wght": 700 if style == "Bold" else 400, **spec.get("fixed_axes", {})}
            static = instancer.instantiateVariableFont(TTFont(io.BytesIO(source)), axes)
            set_static_names(static, family, style)
            path = OUT / f"{ps_family}-{style}.ttf"
            save(static, path)
            faces[weight] = face_entry(path)
    else:
        for weight, filename in spec["static"].items():
            path = OUT / f"{ps_family}-{'Bold' if weight == 'bold' else 'Regular'}.ttf"
            path.write_bytes(fetch(f"{spec['dir']}/{filename}"))
            faces[weight] = face_entry(path)
    license_name = f"{ps_family}-{Path(spec['license']).stem}.txt"
    (OUT / "licenses" / license_name).write_bytes(fetch(f"{spec['dir']}/{spec['license']}"))
    return {"family": family, "aliases": spec["aliases"], "license": f"licenses/{license_name}", **faces}


def main() -> int:
    (OUT / "licenses").mkdir(parents=True, exist_ok=True)
    for old in OUT.glob("*.ttf"):
        old.unlink()
    registry = {
        "source": {"repo": "google/fonts", "commit": COMMIT},
        "default": DEFAULT,
        "families": [build_family(spec) for spec in FAMILIES],
    }
    (OUT / "fonts.json").write_text(json.dumps(registry, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")
    sys.stdout.write(f"Wrote {len(registry['families'])} families to {OUT}\n")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
"""The one list of subtitle fonts (SP4 spec 5.2), read from backend/fonts/fonts.json.

Stdlib only: imported at startup through crop_utils, so it must stay light (E.21).
"""
import json
import os
from functools import lru_cache

FONTS_DIR = os.path.join(os.path.dirname(os.path.abspath(__file__)), "fonts")


@lru_cache(maxsize=1)
def _registry() -> dict:
    with open(os.path.join(FONTS_DIR, "fonts.json"), encoding="utf-8") as f:
        return json.load(f)


@lru_cache(maxsize=1)
def _by_name() -> dict:
    table = {}
    for family in _registry()["families"]:
        table[family["family"].casefold()] = family
        for alias in family.get("aliases", []):
            table[alias.casefold()] = family
    return table


def default_family() -> str:
    return _registry()["default"]


def _entry(name: object) -> dict:
    key = name.strip().casefold() if isinstance(name, str) else ""
    return _by_name().get(key) or _by_name()[default_family().casefold()]


def resolve_font_family(name: object) -> str:
    """Registry family for a stored name: aliases (Arial, Impact, Courier New) and unknown names included."""
    return _entry(name)["family"]


def has_bold(name: object) -> bool:
    return _entry(name).get("bold") is not None


def ass_fontsdir_value() -> str:
    """FONTS_DIR escaped for the `fontsdir` option of FFmpeg's ass filter."""
    return FONTS_DIR.replace("\\", "/").replace(":", "\\\\:").replace("'", "\\\\'")
```

```ts
// The one subtitle font list (SP4 spec 5.2). Same file as the backend: backend/fonts/fonts.json.
import registry from "../../backend/fonts/fonts.json";

interface FontFaceEntry { file: string; postscript: string; sha256: string }
interface FontFamilyEntry {
  family: string;
  aliases: string[];
  license: string;
  regular: FontFaceEntry;
  bold: FontFaceEntry | null;
}
interface FontRegistry { default: string; families: FontFamilyEntry[] }

const REGISTRY = registry as FontRegistry;
const FILE_URLS = import.meta.glob("../../backend/fonts/*.ttf", {
  query: "?url",
  import: "default",
  eager: true,
}) as Record<string, string>;

function entry(name?: string): FontFamilyEntry {
  const key = (name ?? "").trim().toLowerCase();
  const found = REGISTRY.families.find(
    (f) => f.family.toLowerCase() === key || f.aliases.some((a) => a.toLowerCase() === key),
  );
  return found ?? REGISTRY.families.find((f) => f.family === REGISTRY.default)!;
}

export function fontFamilies(): string[] {
  return REGISTRY.families.map((f) => f.family);
}

export function resolveFontFamily(name?: string): string {
  return entry(name).family;
}

export function hasBold(name?: string): boolean {
  return entry(name).bold !== null;
}

export function previewFontFamily(name?: string): string {
  return `"AC ${resolveFontFamily(name)}"`;
}

function faceRule(family: string, face: FontFaceEntry, weight: 400 | 700): string {
  const url = FILE_URLS[`../../backend/fonts/${face.file}`];
  return `@font-face{font-family:"AC ${family}";src:url("${url}") format("truetype");font-weight:${weight};font-style:normal;font-display:swap;}`;
}

export function fontFaceCss(): string {
  return REGISTRY.families
    .flatMap((f) => [faceRule(f.family, f.regular, 400), ...(f.bold ? [faceRule(f.family, f.bold, 700)] : [])])
    .join("\n");
}

export function ensureFontFaces(doc: Document = document): void {
  if (doc.getElementById("ac-subtitle-fonts")) return;
  const style = doc.createElement("style");
  style.id = "ac-subtitle-fonts";
  style.textContent = fontFaceCss();
  doc.head.appendChild(style);
}
```
